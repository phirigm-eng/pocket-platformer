class Level:
    def __init__(self, name, cols, rows=10, sky=((96, 178, 236), (196, 232, 250)),
                 dirt=(122, 86, 60), top=(92, 176, 72), cloud=(255, 255, 255),
                 ice=False, hint=""):
        self.name, self.cols, self.rows = name, cols, rows
        self.sky, self.dirt, self.top, self.cloud = sky, dirt, top, cloud
        self.ice, self.hint = ice, hint
        self.grid = [[0] * cols for _ in range(rows)]
        self.spawn = (2, 2)
        self.goal = (cols - 4, 2)       # tile where the flag pole stands
        self.spikes, self.enemies, self.coins, self.checkpoints = [], [], [], []
        self.block(0, cols - 1, 0, 1)   # ground

    def block(self, c0, c1, t0, t1):
        for ty in range(t0, t1 + 1):
            for tx in range(c0, c1 + 1):
                self.grid[ty][tx] = 1

    def plat(self, c0, c1, ty):
        self.block(c0, c1, ty, ty)

    def pit(self, c0, c1):
        for tx in range(c0, c1 + 1):
            self.grid[0][tx] = self.grid[1][tx] = 0

    def spike(self, *cols, ty=2):
        for c in cols:
            self.spikes.append((c, ty))

    def enemy(self, tx, c0, c1, ty=2):
        self.enemies.append((tx, ty, c0, c1))

    def coin(self, tx, ty):
        self.coins.append((tx, ty))

    def coins_row(self, c0, c1, ty):
        for c in range(c0, c1 + 1):
            self.coin(c, ty)

    def checkpoint(self, *cols, ty=2):
        for c in cols:
            self.checkpoints.append((c, ty))

    def shaft(self, c):
        """Wall-jump shaft: hanging left wall at c, tall right wall at c+4."""
        self.block(c, c, 4, 7)
        self.block(c + 4, c + 4, 2, 7)
        for tx, ty in ((c + 1, 3), (c + 3, 4), (c + 1, 5), (c + 3, 6), (c + 2, 7)):
            self.coin(tx, ty)

    def stairs(self, c, n):
        for i in range(n):
            self.block(c + i, c + i, 2, 2 + i)

def solid(tx, ty):
    if tx < 0 or tx >= COLS:
        return True
    if ty < 0 or ty >= ROWS:

@window.event
def on_resize(width, height):
    """Stretch the logical game view to fill the complete normal window."""
    global view_scale_x, view_scale_y, view_ox, view_oy
    if not width or not height:
        return pyglet.event.EVENT_HANDLED
    view_scale_x = width / WIN_W
    view_scale_y = height / WIN_H
    view_ox = view_oy = 0.0
    try:
        fbw, fbh = window.get_framebuffer_size()
        window.viewport = (0, 0, fbw, fbh)
        if Mat4 is not None:
            window.projection = Mat4.orthogonal_projection(0, WIN_W, 0, WIN_H, -255, 255)
    except Exception:
        pass
    return pyglet.event.EVENT_HANDLED


def to_logical(x, y):
    return ((x - view_ox) / view_scale_x, (y - view_oy) / view_scale_y)

def plus(pair, x, y, w, h, c):
    """A rectangle with chamfered corners, built from two crossing rectangles."""
    c = max(0.0, min(c, w / 2 - 1, h / 2 - 1))
    a, b = pair
    a.x, a.y, a.width, a.height = x + c, y, w - 2 * c, h
    b.x, b.y, b.width, b.height = x, y + c, w, h - 2 * c


def put(s, x, y):
    if s.x != x:
        s.x = x
    if s.y != y:
        s.y = y

class Particle:
    def __init__(self):
        self.s = rect(0, 0, 6, 6, (255, 255, 255), G_PART)
        self.s.visible = False
        self.x = self.y = self.vx = self.vy = self.g = 0.0
        self.life = self.max = 0.0
        self.size = 6


particles = [Particle() for _ in range(110)]
part_i = 0


def emit(x, y, n, color, speed=140, up=120, life=0.5, size=6, gravity=-900):
    global part_i
    for _ in range(n):
        q = particles[part_i]
        part_i = (part_i + 1) % len(particles)
        a = random.uniform(0, 6.283)
        sp = random.uniform(0.3, 1.0) * speed
        q.x, q.y = x, y
        q.vx = math.cos(a) * sp
        q.vy = abs(math.sin(a)) * sp * 0.6 + random.uniform(0, up)
        q.life = q.max = life * random.uniform(0.7, 1.2)
        q.g, q.size = gravity, size
        q.s.color = color
        q.s.width = q.s.height = size
        q.s.visible = True


def update_particles(dt):
    for q in particles:
        if q.life > 0:
            q.life -= dt
            q.vy += q.g * dt
            q.x += q.vx * dt
            q.y += q.vy * dt
            if q.life <= 0:
                q.s.visible = False

class Enemy:
    W, H = 44, 36

    def __init__(self, tx, ty, c0, c1):
        self.start_x = tx * T + 10
        self.x0, self.x1 = c0 * T, (c1 + 1) * T
        self.y = ty * T
        self.phase = random.uniform(0, 6.28)
        pair = lambda col, grp: [rect(0, 0, 4, 4, col, grp), rect(0, 0, 4, 4, col, grp)]
        self.outline = pair(E_OUT, G_E_OUT)
        self.body = pair(E_BODY, G_E_BODY)
        self.hl = rect(0, 0, 4, 4, E_LIGHT, G_E_BODY)
        self.belly = rect(0, 0, 4, 4, E_BELLY, G_E_BODY)
        self.feet = [rect(0, 0, 12, 6, E_OUT, G_E_OUT) for _ in range(2)]
        self.eyes = [rect(0, 0, 11, 13, (255, 255, 255), G_E_FACE) for _ in range(2)]
        self.pupils = [rect(0, 0, 5, 7, (30, 20, 40), G_E_FACE) for _ in range(2)]
        self.brows = [rect(0, 0, 7, 4, E_OUT, G_E_FACE) for _ in range(4)]
        self.face = self.eyes + self.pupils + self.brows + self.feet + [self.hl, self.belly]
        self.reset()

    def reset(self):
        self.x = self.start_x
        self.dir = 1
        self.alive = True
        self.squish = 0.0

    def update(self, dt):
        if not self.alive:
            self.squish = max(0.0, self.squish - dt)
            return
        self.x += self.dir * 70 * dt
        if self.x < self.x0:
            self.x, self.dir = self.x0, 1
        elif self.x + self.W > self.x1:
            self.x, self.dir = self.x1 - self.W, -1

    def all_shapes(self):
        return self.outline + self.body + self.face

    def destroy(self):
        for s in self.all_shapes():
            s.delete()

    def layout(self, cx, cy, t):
        parts = self.all_shapes()
        visible = self.alive or self.squish > 0
        for s in parts:
            s.visible = visible
        if not visible:
            return
        ph = t * 7 + self.phase
        if self.alive:
            w = self.W * (1 + 0.05 * math.sin(ph + 1.2))
            h = (self.H - 4) * (1 + 0.07 * math.sin(ph))
            y = self.y - cy + 4
        else:
            w, h = self.W * 1.25, 10
            y = self.y - cy
        x = self.x + self.W / 2 - w / 2 - cx
        x, y = int(x), int(y)
        plus(self.outline, x - 3, y - 3, w + 6, h + 6, 10)
        plus(self.body, x, y, w, h, 8)
        if not self.alive:
            for s in self.face:
                s.visible = False
            return
        self.hl.x, self.hl.y, self.hl.width, self.hl.height = x + 9, y + h - 7, w - 18, 3
        self.belly.x, self.belly.y, self.belly.width, self.belly.height = x + w * 0.2, y + 3, w * 0.6, 9
        # feet shuffle
        for k, fx in enumerate((x + 4, x + w - 16)):
            lift = max(0.0, math.sin(ph * 1.4 + k * 3.14)) * 3
            put(self.feet[k], fx, y - 4 + lift)
        # angry face
        ey = y + h * 0.40
        mid = x + w / 2
        for k, ex in enumerate((mid - 14, mid + 3)):
            put(self.eyes[k], ex, ey)
            put(self.pupils[k], ex + (5 if self.dir > 0 else 1), ey + 3)
        e0, e1 = mid - 14, mid + 3
        # left eye: outer end high, inner end low; right eye mirrored
        put(self.brows[0], e0 - 1, ey + 15)
        put(self.brows[1], e0 + 6, ey + 12)
        put(self.brows[2], e1 + 5, ey + 15)
        put(self.brows[3], e1 - 2, ey + 12)


class Player:
    W, H = 40, 40

    def __init__(self):
        pair = lambda col, grp: [rect(0, 0, 4, 4, col, grp), rect(0, 0, 4, 4, col, grp)]
        self.outline = pair(P_OUT, G_P_OUT)
        self.body = pair(P_BODY, G_P_BODY)
        self.light = rect(0, 0, 4, 4, P_LIGHT, G_P_DET)
        self.shade = rect(0, 0, 4, 4, P_SHADE, G_P_DET)
        self.band = rect(0, 0, 4, 4, P_BAND, G_P_DET)
        self.tails = [rect(0, 0, 12, 5, P_BAND, G_P_DET), rect(0, 0, 9, 5, P_BAND, G_P_DET)]
        self.eyes = [rect(0, 0, 9, 12, (255, 255, 255), G_P_FACE) for _ in range(2)]
        self.pupils = [rect(0, 0, 5, 6, (30, 20, 40), G_P_FACE) for _ in range(2)]
        self.parts = (self.outline + self.body + [self.light, self.shade, self.band]
                      + self.tails + self.eyes + self.pupils)
        self.sx = self.sy = 1.0
        self.t = 0.0
        self.next_blink = 2.0
        self.blinking = 0.0
        self.reset()

    def reset(self):
        self.x, self.y = spawn_pos
        self.vx = self.vy = 0.0
        self.facing = 1
        self.on_ground = False
        self.wall = 0
        self.sliding = False
        self.coyote = 0.0
        self.jump_buf = 0.0
        self.can_double = True
        self.lock_t = 0.0
        self.hit_t = 0.0
        self.dj_t = 0.0
        self.invuln = 1.0
        self.squash = 0.0
        self.stretch = 0.0
        self.slide_dust = 0.0

    def layout(self, cx, cy, dt):
        self.t += dt
        # squash & stretch, idle breathing, run bob
        tsx = tsy = 1.0
        if self.squash > 0:
            k = self.squash / 0.16
            tsy, tsx = 1 - 0.30 * k, 1 + 0.30 * k
        elif self.stretch > 0:
            k = self.stretch / 0.14
            tsy, tsx = 1 + 0.22 * k, 1 - 0.16 * k
        elif not self.on_ground:
            s = min(abs(self.vy) / 1000, 1) * 0.14
            tsy, tsx = 1 + s, 1 - s * 0.7
        elif abs(self.vx) > 25:
            tsy += 0.05 * math.sin(self.t * 22)
        else:
            tsy += 0.025 * math.sin(self.t * 3.2)
        k = min(1.0, 22 * dt)
        self.sx += (tsx - self.sx) * k
        self.sy += (tsy - self.sy) * k

        w, h = self.W * self.sx, self.H * self.sy
        x = int(self.x + self.W / 2 - w / 2 - cx)
        y = int(self.y - cy)
        f = self.facing
        plus(self.outline, x - 3, y - 3, w + 6, h + 6, 9)
        plus(self.body, x, y, w, h, 7)
        self.light.x, self.light.y, self.light.width, self.light.height = x + 8, y + h - 7, w - 16, 3
        self.shade.x, self.shade.y, self.shade.width, self.shade.height = x + 8, y + 2, w - 16, 5

        # headband with fluttering tails
        by = y + h * 0.60
        self.band.x, self.band.y, self.band.width, self.band.height = x, by, w, 7
        wig = math.sin(self.t * 16) * 3
        trail = -max(-1.0, min(1.0, self.vx / RUN_SPEED)) * 6
        if f > 0:
            put(self.tails[0], x - 11 + min(0, trail), by + 1 + wig)
            put(self.tails[1], x - 8 + min(0, trail), by - 5 - wig)
        else:
            put(self.tails[0], x + w - 1 + max(0, trail), by + 1 + wig)
            put(self.tails[1], x + w - 1 + max(0, trail), by - 5 - wig)

        # eyes: look where we move, blink now and then, squint when hurt
        self.next_blink -= dt
        if self.next_blink < 0:
            self.blinking = 0.12
            self.next_blink = random.uniform(2.0, 4.5)
        self.blinking = max(0.0, self.blinking - dt)
        shut = self.blinking > 0 or self.hit_t > 0
        eh = 3 if shut else 12
        ex = x + w / 2 + f * 5
        ey = y + h * 0.22
        look = 2 if self.vy > 250 else (-2 if self.vy < -250 else 0)
        for k, off in enumerate((-11, 2)):
            e = self.eyes[k]
            e.x, e.y, e.height = ex + off, ey, eh
            pu = self.pupils[k]
            pu.visible = not shut
            pu.x, pu.y = ex + off + 2 + f * 2, ey + 3 + look

        blink = self.invuln > 0 and int(self.invuln * 14) % 2 == 0
        op = 110 if blink else 255
        for s in self.parts:
            if s.opacity != op:
                s.opacity = op

def approach(v, target, amount):
    if v < target:
        return min(v + amount, target)
    return max(v - amount, target)


def wall_probe(p, side):
    px = p.x - 2 if side < 0 else p.x + p.W + 2
    return solid_at(px, p.y + 6) or solid_at(px, p.y + p.H - 6)


def overlaps(ax, ay, aw, ah, bx, by, bw, bh):
    return ax < bx + bw and ax + aw > bx and ay < by + bh and ay + ah > by


def game_over():
    global state

    overlay.opacity = 140
    title_label.text = "GAME OVER"
    sub_label.text = "press R or tap the screen to try again"


def hurt(p, from_x):
    global lives, shake
    if p.invuln > 0 or state != "play":
        return
    lives -= 1
    shake = 0.3

    p.invuln = 1.6

    p.hit_t = 0.0
    emit(p.x + p.W / 2, p.y, 10, (255, 255, 255), speed=180, up=160, life=0.5)

def step(dt):
    global state, elapsed, banner_t, shake, spawn_pos, new_best
    p = player
    playing = state == "play"
    if playing:
        elapsed += dt
        banner_t = max(0.0, banner_t - dt)

    left = control_held("left")
    right = control_held("right")
    inp = (1 if right else 0) - (1 if left else 0)
    if inp == 0:
        inp = run_latch
    held = control_held("jump") or touch_jump_held
    if not playing or p.hit_t > 0:
        inp = 0
        p.jump_buf = 0

    p.hit_t = max(0.0, p.hit_t - dt)
    p.invuln = max(0.0, p.invuln - dt)
    p.dj_t = max(0.0, p.dj_t - dt)
    p.squash = max(0.0, p.squash - dt)
    p.stretch = max(0.0, p.stretch - dt)
    p.slide_dust = max(0.0, p.slide_dust - dt)
    p.jump_buf = max(0.0, p.jump_buf - dt)

    # contact with the world
    if p.on_ground:
        p.coyote = COYOTE
        p.can_double = True
    else:
        p.coyote -= dt
    p.wall = 0
    if not p.on_ground:
        if wall_probe(p, -1):
            p.wall = -1
        elif wall_probe(p, 1):
            p.wall = 1
    p.sliding = (p.wall != 0 and not p.on_ground and p.vy <= 0
                 and inp == p.wall and p.hit_t <= 0)
    if p.sliding and p.slide_dust <= 0:
        p.slide_dust = 0.07
        emit(p.x + (0 if p.wall < 0 else p.W), p.y + p.H * 0.5, 1, (235, 225, 210),
             speed=40, up=20, life=0.3, size=4, gravity=-200)

    # horizontal movement
    ground_accel, ground_fric = ((ICE_ACCEL, ICE_FRICTION) if level.ice
                                 else (GROUND_ACCEL, GROUND_FRICTION))
    if p.lock_t > 0:
        p.lock_t -= dt
    elif inp != 0:
        accel = ground_accel if p.on_ground else AIR_ACCEL
        if (inp > 0) != (p.vx > 0) and p.vx != 0:
            accel *= 1.5
        p.vx = approach(p.vx, inp * RUN_SPEED, accel * dt)
        p.facing = inp
    elif p.hit_t <= 0:
        fr = ground_fric if p.on_ground else AIR_FRICTION
        p.vx = approach(p.vx, 0, fr * dt)

    # jumping
    dust = (235, 225, 210)
    if p.jump_buf > 0:
        if p.wall != 0 and not p.on_ground and p.coyote <= 0:
            p.vy = WALL_JUMP_V
            p.vx = -p.wall * WALL_JUMP_PUSH
            p.facing = -p.wall
            p.lock_t = 0.16
            p.can_double = True
            p.jump_buf = 0
            p.stretch = 0.14
            emit(p.x + (0 if p.wall < 0 else p.W), p.y + p.H * 0.4, 5, dust, speed=110,
                 up=60, life=0.35, size=5, gravity=-300)
        elif p.coyote > 0:
            p.vy = JUMP_V
            p.coyote = 0
            p.on_ground = False
            p.jump_buf = 0
            p.stretch = 0.14
            emit(p.x + p.W / 2, p.y, 5, dust, speed=120, up=30, life=0.35, size=5, gravity=-300)
        elif p.can_double:
            p.vy = DOUBLE_JUMP_V
            p.can_double = False
            p.dj_t = 0.4
            p.jump_buf = 0
            p.stretch = 0.14
            emit(p.x + p.W / 2, p.y + 4, 9, (255, 255, 255), speed=170, up=20, life=0.4,
                 size=5, gravity=-200)

    # gravity (short hop if the jump button is released early)
    g = GRAVITY
    if p.vy > 0 and not held:
        g *= 2.4
    p.vy = max(p.vy + g * dt, MAX_FALL)
    if p.sliding:
        p.vy = max(p.vy, WALL_SLIDE_SPEED)

    # move X and resolve
    p.x += p.vx * dt
    for ty in range(math.floor(p.y / T), math.floor((p.y + p.H - 0.01) / T) + 1):
        for tx in range(math.floor(p.x / T), math.floor((p.x + p.W - 0.01) / T) + 1):
            if solid(tx, ty):
                if p.vx > 0:
                    p.x = tx * T - p.W
                elif p.vx < 0:
                    p.x = (tx + 1) * T
                p.vx = 0.0

    # move Y and resolve
    fall_v = p.vy
    was_ground = p.on_ground
    p.y += p.vy * dt
    p.on_ground = False
    for tx in range(math.floor(p.x / T), math.floor((p.x + p.W - 0.01) / T) + 1):
        for ty in range(math.floor(p.y / T), math.floor((p.y + p.H - 0.01) / T) + 1):
            if solid(tx, ty):
                if p.vy <= 0:
                    p.y = (ty + 1) * T
                    p.on_ground = True
                else:
                    p.y = ty * T - p.H
                p.vy = 0.0
    if p.on_ground and not was_ground and fall_v < -260:
        p.squash = 0.16
        emit(p.x + 6, p.y, 4, dust, speed=110, up=30, life=0.35, size=5, gravity=-300)
        emit(p.x + p.W - 6, p.y, 4, dust, speed=110, up=30, life=0.35, size=5, gravity=-300)

    for e in enemies:
        e.update(dt)

    if state != "play":
        return

    if p.y < -T:
        pit_fall(p)
        return

    pcx, pcy = p.x + p.W / 2, p.y + p.H / 2
    for c in coins:
        if not c.taken and abs(c.cx - pcx) < 26 and abs(c.cy - pcy) < 34:
            c.taken = True
            add_score(100)
            emit(c.cx, c.cy, 6, (255, 226, 110), speed=150, up=100, life=0.45, size=5, gravity=-500)

    for cp in checkpoints:
        if not cp.active and overlaps(p.x, p.y, p.W, p.H, cp.x, cp.y, T, 2 * T):
            cp.active = True
            add_score(250)
            cp.flag.color = (70, 204, 112)
            spawn_pos = (cp.x + 12, cp.y)
            emit(cp.x + 32, cp.y + 90, 10, (120, 240, 150), speed=140, up=120, life=0.6, size=5,
                 gravity=-300)

    for s in spikes:
        bx, by, bw, bh = s.box()
        if overlaps(p.x + 4, p.y, p.W - 8, p.H - 6, bx, by, bw, bh):
            hurt(p, bx + bw / 2)

    for e in enemies:
        if e.alive and overlaps(p.x, p.y, p.W, p.H, e.x, e.y, e.W, e.H):
            if p.vy < 0 and p.y > e.y + e.H * 0.45:
                e.alive = False
                e.squish = 0.4
                add_score(200)
                p.vy = 640
                p.can_double = True
                p.stretch = 0.14
                shake = max(shake, 0.12)
                emit(e.x + e.W / 2, e.y + 8, 12, E_BODY, speed=220, up=160, life=0.55, size=6)
            else:
                hurt(p, e.x + e.W / 2)

    if overlaps(p.x, p.y, p.W, p.H, goal_x, goal_y, T, 3 * T):
        state = "win"
        add_score(1000)
        overlay.opacity = 120
        got = sum(1 for c in coins if c.taken)
        key_ = str(level_i)
        old = best_times.get(key_)
        new_best = old is None or elapsed < old
        if new_best:
            best_times[key_] = round(elapsed, 2)
            save_best()
        emit(pcx, pcy, 24, (255, 226, 110), speed=300, up=260, life=0.9, size=7)
        if level_i == len(LEVEL_BUILDERS) - 1:
            title_label.text = "YOU BEAT THE GAME!"
            tail = "press 1 or tap to start over"
        else:
            title_label.text = "LEVEL CLEAR!"
            tail = "press Enter or tap for the next level"
        best_txt = "NEW BEST!" if new_best else "best %.1fs" % old
        sub_label.text = "%d / %d coins   %.1fs   %s\n%s" % (got, len(coins), elapsed, best_txt, tail)


# --------------------------------------------------------------------------
# Rendering
# --------------------------------------------------------------------------
def render(dt):
    global cam, cam_y, shake, tiles_dirty, last_cam
    p = player

    tx, ty = target_cam()
    k = min(1.0, 7 * dt)
    cam += (tx - cam) * k
    cam_y += (ty - cam_y) * k
    shake = max(0.0, shake - dt)
    sx_, sy_ = 0, 0
    if shake > 0:
        sx_ = int(random.uniform(-1, 1) * shake * 30)
        sy_ = int(random.uniform(-1, 1) * shake * 30)
    cx, cy = int(round(cam)) + sx_, int(round(cam_y)) + sy_

    # hills (parallax)
    for hl in hill_layers:
        off = cam * hl["par"]
        first = int(off // HILL_W)
        y0 = -cam_y * hl["par"] * 0.6 - 40
        for i, s in enumerate(hl["shapes"]):
            wx = (first + i) * HILL_W
            h = (hl["base"] + hl["a1"] * math.sin(wx * hl["f1"] + hl["ph"])
                 + hl["a2"] * math.sin(wx * hl["f2"] + 1.3 * hl["ph"]))
            s.x, s.y, s.height = wx - off, y0, h + 40

    # clouds
    span = 9 * 330
    for x, y, parts in clouds:
        sx = (x - cam * 0.25) % span - 160
        sy = y - cam_y * 0.15
        parts[0].x, parts[0].y = sx, sy
        parts[1].x, parts[1].y = sx + 20, sy + 30
        parts[2].x, parts[2].y = sx + 70, sy + 56

    # tiles (only touched when the camera moved)
    if tiles_dirty or last_cam != (cx, cy):
        tiles_dirty = False
        last_cam = (cx, cy)
        first_c, first_r = cx // T, cy // T
        for i in range(VIS_COLS):
            c = first_c + i
            sx = c * T - cx
            for j in range(VIS_ROWS):
                t = first_r + j
                idx = i * VIS_ROWS + j
                if 0 <= c < COLS and 0 <= t < ROWS and grid[t][c]:
                    body = tile_body[idx]
                    body.visible = True
                    put(body, sx, t * T - cy)
                    strip = tile_top[idx]
                    if not solid(c, t + 1):
                        strip.visible = True
                        put(strip, sx, t * T - cy + T - 14)
                    else:
                        strip.visible = False
                else:
                    tile_body[idx].visible = False
                    tile_top[idx].visible = False

    # coins
    for c in coins:
        vis = not c.taken
        c.rim.visible = c.outer.visible = c.inner.visible = vis
        if vis:
            bob = math.sin(elapsed * 4 + c.phase) * 5
            x = int(c.cx - cx)
            y = int(c.cy + bob - cy)
            for s in (c.rim, c.outer, c.inner):
                s.x, s.y = x, y

    # spikes
    for s in spikes:
        x, y = s.x - cx, s.y - cy
        for part, ox, oy in zip(s.outline, (4, 12, 21), (-1, 11, 23)):
            put(part, x + ox, y + oy)
        for part, ox, oy in zip(s.parts, (6, 14, 23), (0, 12, 24)):
            put(part, x + ox, y + oy)

    # checkpoints
    for cp in checkpoints:
        put(cp.pole, cp.x + 29 - cx, cp.y - cy)
        put(cp.flag, cp.x + 35 - cx, cp.y + 78 - cy + (math.sin(elapsed * 5) * 2 if cp.active else 0))

    # enemies
    for e in enemies:
        e.layout(cx, cy, elapsed)

    # particles
    for q in particles:
        if q.life > 0:
            q.s.x, q.s.y = q.x - cx - q.size / 2, q.y - cy
            q.s.opacity = int(255 * max(0.0, q.life / q.max))

    # goal flag
    gx, gy = goal_x - cx, goal_y - cy
    goal_pole.x, goal_pole.y = gx + 28, gy
    wave = math.sin(elapsed * 6) * 3
    for j, part in enumerate(goal_flag):
        part.x = gx + 36 + (wave if j == 2 else 0)
        part.y = gy + 3 * T - 18 * (j + 1)

    # player
    p.layout(cx, cy, dt)

    # HUD
    got = sum(1 for c in coins if c.taken)
    set_text(coin_label, "LEVEL %d/%d   COINS %d/%d" % (level_i + 1, len(LEVEL_BUILDERS), got, len(coins)))
    set_text(time_label, "%.1f s" % elapsed)
    set_text(score_label, "SCORE %d   BEST %d" % (score, high_score))
    for i, h in enumerate(hearts):
        for s in h:
            s.visible = i < lives
    if state == "play" and not paused:
        set_text(title_label, ("LEVEL %d  -  %s" % (level_i + 1, level.name)) if banner_t > 0 else "")
        if elapsed > 8:
            set_text(hint_label, "")
    if show_touch:
        for name, r in btn_rect.items():
            held = (name == "jump" and touch_jump_held)
            latched = (name == "left" and run_latch < 0) or (name == "right" and run_latch > 0)
            op = 120 if held else (90 if latched else 50)
            if r.opacity != op:
                r.opacity = op


def update(dt):
    global acc
    update_particles(dt)
    if paused:
        render(dt)
        return
    acc += min(dt, 0.05)
    while acc >= STEP:
        step(STEP)
        acc -= STEP
    render(dt)

LEVEL_KEYS = (key._1, key._2, key._3, key._4, key._5)
MOVE_KEYS = CONTROL_KEYS["left"] + CONTROL_KEYS["right"]


def set_paused(v):
    global paused
    if state != "play":
        return
    paused = v
    if v:
        overlay.opacity = 120
        title_label.text = "PAUSED"
        sub_label.text = "press P or tap to resume"
    else:
        overlay.opacity = 0
        title_label.text = ""
        sub_label.text = ""


def advance():
    """Continue after a win / game over (Enter, or a tap on touch screens)."""
    if state == "win":
        load_level(level_i + 1 if level_i < len(LEVEL_BUILDERS) - 1 else 0)
    elif state == "dead":
        load_level(level_i)


@window.event
def on_key_press(symbol, modifiers):
    global run_latch
    keys.add(symbol)
    if state == "menu":
        if symbol in (key.ENTER, key.RETURN, key.SPACE):
            start_game()
        return
    if symbol in MOVE_KEYS:
        run_latch = 0
    if symbol in CONTROL_KEYS["jump"]:
        player.jump_buf = JUMP_BUFFER
    elif symbol == key.R:
        load_level(level_i)
    elif symbol == key.P:
        set_paused(not paused)
    elif symbol == key.F:
        window.set_fullscreen(not window.fullscreen)
    elif symbol == key.T:
        set_touch_visible(not show_touch)
    elif symbol in (key.ENTER, key.RETURN) and state == "win":
        if level_i < len(LEVEL_BUILDERS) - 1:
            load_level(level_i + 1)
    elif symbol in LEVEL_KEYS:
        load_level(LEVEL_KEYS.index(symbol))


@window.event
def on_key_release(symbol, modifiers):
    keys.discard(symbol)


@window.event
def on_deactivate():
    global touch_jump_held
    keys.clear()
    touch_jump_held = False
    if state == "play":
        set_paused(True)


def hit(name, lx, ly):
    bx, by, bw, bh = BTN[name]
    return bx <= lx <= bx + bw and by <= ly <= by + bh


@window.event
def on_mouse_press(x, y, button, modifiers):
    global run_latch, touch_jump_held
    lx, ly = to_logical(x, y)
    if state == "menu":
        start_game()
        return
    if paused:
        set_paused(False)
        return
    if state in ("win", "dead"):
        advance()
        return
    if show_touch and hit("pause", lx, ly):
        set_paused(True)
        return
    if show_touch and hit("left", lx, ly):
        run_latch = 0 if run_latch < 0 else -1
    elif show_touch and hit("right", lx, ly):
        run_latch = 0 if run_latch > 0 else 1
    elif show_touch and hit("jump", lx, ly):
        player.jump_buf = JUMP_BUFFER
        touch_jump_held = True
    elif show_touch:
        # On mobile, tapping anywhere that is not a control is a jump.
        player.jump_buf = JUMP_BUFFER
        touch_jump_held = True


@window.event
def on_mouse_release(x, y, button, modifiers):
    global touch_jump_held
    touch_jump_held = False


@window.event
def on_draw():
    window.clear()
    batch.draw()
    hud_batch.draw()


if __name__ == "__main__":
    on_resize(window.width, window.height)
    load_level(0)
    show_main_menu()
    pyglet.clock.schedule_interval(update, 1 / 60)
    pyglet.app.run()
